// BLS Term 4 Punctuality Pledge — backend // Deploy as Web App (Execute as: Me, Who has access: Anyone), then paste the // resulting /exec URL into CONFIG.WEBAPP_URL in both student-pledge-form.html // and teacher-pledge-dashboard.html. var SHEET_SUGGESTIONS = 'Suggestions'; var SHEET_VOTES = 'Votes'; var SHEET_PLEDGES = 'Pledges'; // Change this before deploying. Required for any data-clearing action // (per-CCA clear and whole-school reset). This is a soft deterrent, not // real security — anyone who knows it (or intercepts a request) can use // it, since this script has no login system. var ADMIN_PASSWORD = 'BlsSMT2026!'; function getSheet_(name, headers) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName(name); if (!sheet) { sheet = ss.insertSheet(name); sheet.appendRow(headers); } return sheet; } function suggestionsSheet_() { return getSheet_(SHEET_SUGGESTIONS, ['Timestamp', 'SessionId', 'Name', 'CCA', 'Trigger', 'Minutes', 'Reason', 'Sentence', 'Key']); } function votesSheet_() { return getSheet_(SHEET_VOTES, ['Timestamp', 'SessionId', 'Name', 'CCA', 'Key']); } function pledgesSheet_() { return getSheet_(SHEET_PLEDGES, ['CCA', 'Sentence', 'ConfirmedAt']); } function normalizeKey_(sentence) { return String(sentence).trim().toLowerCase().replace(/\s+/g, ' '); } function doPost(e) { var lock = LockService.getScriptLock(); lock.waitLock(10000); try { var body = JSON.parse(e.postData.contents); var action = body.action; var result; if (action === 'submitSuggestion') result = handleSubmitSuggestion_(body); else if (action === 'submitVotes') result = handleSubmitVotes_(body); else if (action === 'resetSession') result = handleResetSession_(body); else if (action === 'confirmPledge') result = handleConfirmPledge_(body); else if (action === 'clearAllCCA') result = handleClearAllCCA_(body); else if (action === 'clearAllSchool') result = handleClearAllSchool_(body); else result = { ok: false, error: 'Unknown action' }; return outputJson_(result); } catch (err) { return outputJson_({ ok: false, error: String(err) }); } finally { lock.releaseLock(); } } function handleSubmitSuggestion_(body) { var sheet = suggestionsSheet_(); var key = normalizeKey_(body.sentence); sheet.appendRow([new Date(), body.sessionId, body.name, body.cca, body.trigger, body.minutes, body.reason, body.sentence, key]); return { ok: true, key: key }; } function handleSubmitVotes_(body) { var sheet = votesSheet_(); var keys = body.keys || []; keys.forEach(function (k) { sheet.appendRow([new Date(), body.sessionId, body.name, body.cca, k]); }); return { ok: true }; } function handleResetSession_(body) { deleteRowsByValue_(suggestionsSheet_(), body.sessionId, 1); deleteRowsByValue_(votesSheet_(), body.sessionId, 1); return { ok: true }; } function deleteRowsByValue_(sheet, value, colIndex) { var data = sheet.getDataRange().getValues(); for (var i = data.length - 1; i >= 1; i--) { if (data[i][colIndex] === value) { sheet.deleteRow(i + 1); } } } function handleConfirmPledge_(body) { var sheet = pledgesSheet_(); var data = sheet.getDataRange().getValues(); var rowIndex = -1; for (var i = 1; i < data.length; i++) { if (data[i][0] === body.cca) { rowIndex = i + 1; break; } } var row = [body.cca, body.sentence, new Date()]; if (rowIndex > -1) sheet.getRange(rowIndex, 1, 1, 3).setValues([row]); else sheet.appendRow(row); return { ok: true }; } function handleClearAllCCA_(body) { if (body.password !== ADMIN_PASSWORD) return { ok: false, error: 'Incorrect password' }; deleteRowsByValue_(suggestionsSheet_(), body.cca, 3); deleteRowsByValue_(votesSheet_(), body.cca, 3); deleteRowsByValue_(pledgesSheet_(), body.cca, 0); return { ok: true }; } function handleClearAllSchool_(body) { if (body.password !== ADMIN_PASSWORD) return { ok: false, error: 'Incorrect password' }; clearAllRows_(suggestionsSheet_()); clearAllRows_(votesSheet_()); clearAllRows_(pledgesSheet_()); return { ok: true }; } function clearAllRows_(sheet) { var lastRow = sheet.getLastRow(); if (lastRow > 1) sheet.deleteRows(2, lastRow - 1); } function doGet(e) { var action = e.parameter.action; if (action === 'state') return outputJson_(getState_(e.parameter.cca)); if (action === 'overview') return outputJson_(getOverview_()); return outputJson_({ ok: false, error: 'Unknown action' }); } function outputJson_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } // Merged, vote-counted suggestion list for one CCA, plus its confirmed pledge if any. function getState_(cca) { var sData = suggestionsSheet_().getDataRange().getValues(); var vData = votesSheet_().getDataRange().getValues(); var pData = pledgesSheet_().getDataRange().getValues(); var byKey = {}; for (var i = 1; i < sData.length; i++) { var srow = sData[i]; if (srow[3] !== cca) continue; var key = srow[8]; if (!byKey[key]) byKey[key] = { key: key, sentence: srow[7], votes: 0, authors: [] }; byKey[key].authors.push(srow[2]); } for (var j = 1; j < vData.length; j++) { var vrow = vData[j]; if (vrow[3] !== cca) continue; var vkey = vrow[4]; if (byKey[vkey]) byKey[vkey].votes += 1; } var suggestions = Object.keys(byKey).map(function (k) { return byKey[k]; }); suggestions.sort(function (a, b) { return b.votes - a.votes; }); var pledge = null; for (var p = 1; p < pData.length; p++) { if (pData[p][0] === cca) { pledge = { sentence: pData[p][1], confirmedAt: pData[p][2] }; break; } } return { ok: true, suggestions: suggestions, pledge: pledge }; } // Per-CCA participation counts and confirmed pledges, for the teacher overview tab. function getOverview_() { var sData = suggestionsSheet_().getDataRange().getValues(); var vData = votesSheet_().getDataRange().getValues(); var pData = pledgesSheet_().getDataRange().getValues(); var counts = {}; for (var i = 1; i < sData.length; i++) { var cca = sData[i][3]; counts[cca] = counts[cca] || { suggestions: 0, votes: 0 }; counts[cca].suggestions++; } for (var j = 1; j < vData.length; j++) { var vcca = vData[j][3]; counts[vcca] = counts[vcca] || { suggestions: 0, votes: 0 }; counts[vcca].votes++; } var pledges = {}; for (var p = 1; p < pData.length; p++) { pledges[pData[p][0]] = { sentence: pData[p][1], confirmedAt: pData[p][2] }; } return { ok: true, counts: counts, pledges: pledges }; }
